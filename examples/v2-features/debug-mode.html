Configuration Reference

Debug Mode

windrunner({
  debug: true,
  // Logs cache hits, compile times,
  // failures to console.
  // Exposes window.__WINDRUNNER__
  // Enables extended getStats() data
});

Strict Mode

windrunner({
  strict: true,
  // Throws errors on failed compilation
  // instead of silently skipping.
  // Great for development/CI to catch
  // typos in class names early.
});

Error Callbacks

windrunner({
  onError: (className, context) => {
    // context.reason: 'parse-error',
    //   'unknown-utility', 'unknown-variant',
    //   'compilation-error', 'lite-mode-excluded'
    // context.rawClass: original class string
    // context.variant: failed variant (if any)
  },
  onWarning: (message, context) => {
    // context.severity: 'low'|'medium'|'high'
    // context.timestamp: when it occurred
  }
});

__WINDRUNNER__ Global

// Available when debug: true
window.__WINDRUNNER__ = {
  version: '2.0.0',
  instance: {
    getStats(),
    getCacheSize(),
    getInsertedRuleCount(),
    clearCache(),
    scan(element),
    disconnect()
  }
};
