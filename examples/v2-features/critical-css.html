Node.js Usage (SSR / Build Scripts)

Static Site Generation

// build-critical-css.js (Node.js script)
import { compileCriticalCssFromFiles } from 'windrunner/compiler';
import { writeFileSync } from 'fs';

// Scan source files and generate critical CSS
const css = await compileCriticalCssFromFiles([
  './src/components/Header.jsx',
  './src/components/Hero.jsx',
  './src/pages/index.html',
]);

// Write to dist for inclusion in HTML
writeFileSync('./dist/critical.css', css);

Server-Side Rendering

// server.js (Express/Fastify SSR)
import { compileCriticalCss } from 'windrunner/compiler';

app.get('/', (req, res) => {
  const html = renderApp();  // Your React/Vue render
  
  // Extract classes used in rendered HTML
  const classes = extractClassesFromHtml(html);
  const criticalCss = compileCriticalCss(classes);
  
  res.send(\`
    <!DOCTYPE html>
    <html>
    <head>
      <style>\${criticalCss}</style>
    </head>
    <body>\${html}</body>
    </html>
  \`);
});
