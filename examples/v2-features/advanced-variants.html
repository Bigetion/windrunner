Syntax Reference

has-* variants

has-[:checked]:bg-green-100
has-[:focus]:ring-2
has-[:disabled]:opacity-50
group-has-[:focus]:border-blue-500
peer-has-[:checked]:text-green-500

data-* variants

data-[state=open]:block
data-[state=closed]:hidden
data-[active]:bg-blue-500
data-[size=lg]:text-xl
data-[theme=dark]:bg-slate-900

aria-* variants

aria-[expanded=true]:rotate-180
aria-[selected=true]:bg-blue-500
aria-[disabled=true]:opacity-50
aria-[hidden=true]:hidden
aria-[current=page]:font-bold

Combinations

md:has-[:checked]:bg-blue-500
hover:data-[state=open]:bg-blue-600
group-hover:aria-[expanded=true]:scale-105
data-[state=open]:!block
